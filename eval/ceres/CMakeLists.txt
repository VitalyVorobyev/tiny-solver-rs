cmake_minimum_required(VERSION 3.16)
project(ceres_fit CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(Ceres REQUIRED)
add_executable(robust_fit robust_fit.cc)
target_link_libraries(robust_fit Ceres::ceres)
add_executable(sweep sweep.cc)
target_link_libraries(sweep Ceres::ceres)
add_executable(pgo pgo.cc)
target_link_libraries(pgo Ceres::ceres)
